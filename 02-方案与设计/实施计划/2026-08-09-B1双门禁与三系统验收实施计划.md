```ts
const standardAccessLifetimeMilliseconds = 2 * 60 * 60 * 1000;

function acceptanceConfig(env: ConfigEnvironment, nodeEnv: ApiConfig["nodeEnv"]) {
  const enabled = env.ACCEPTANCE_MODE === "1";
  if (env.ACCEPTANCE_MODE && !["0", "1"].includes(env.ACCEPTANCE_MODE)) {
    invalid("INVALID_ACCEPTANCE_MODE");
  }
  if (!enabled && env.ACCEPTANCE_ACCESS_TTL_SECONDS) {
    invalid("ACCEPTANCE_TTL_WITHOUT_MODE");
  }
  if (!enabled) {
    return { acceptanceMode: false, accessTokenLifetimeMilliseconds: standardAccessLifetimeMilliseconds };
  }
  if (nodeEnv === "production") invalid("ACCEPTANCE_MODE_FORBIDDEN_IN_PRODUCTION");
  const seconds = Number(required(env, "ACCEPTANCE_ACCESS_TTL_SECONDS"));
  if (!Number.isInteger(seconds) || seconds < 30 || seconds > 600) {
    invalid("INVALID_ACCEPTANCE_ACCESS_TTL_SECONDS");
  }
  return { acceptanceMode: true, accessTokenLifetimeMilliseconds: seconds * 1000 };
}
```

`host` 使用：

host: nodeEnv === "production" || acceptance.acceptanceMode

- [ ] **Step 3: 实现生成器校验和原子写入**

`buildRuntimeConfig()` 使用以下边界函数，不得依赖 DNS 结果猜测私网：

```js
function isPrivateIpv4(hostname) {
  const parts = hostname.split(".").map(Number);
  if (parts.length !== 4 || parts.some((part) =>
    !Number.isInteger(part) || part < 0 || part > 255
  )) return false;
  return parts[0] === 10 ||
    (parts[0] === 172 && parts[1] >= 16 && parts[1] <= 31) ||
    (parts[0] === 192 && parts[1] === 168);
}

function hasOriginOnly(parsed) {
  return !parsed.username && !parsed.password &&
    parsed.pathname === "/" && !parsed.search && !parsed.hash;
}

// local 固定 http://127.0.0.1:3100，不接受覆盖。
// acceptance 接受 HTTPS，或三个 RFC1918 范围的 HTTP origin。
// production 只接受 HTTPS origin。
return { mode, apiBase: parsed.origin };
```

命令模式必须为：`--local` 强制写本地配置；`--from-env` 要求两个环境变量并覆盖；
`--ensure` 只在目标不存在时写本地配置。直接运行脚本时把以下确定性内容写入临时
文件，再用 `renameSync` 原子替换目标：

```ts

export const runtimeConfig = Object.freeze({
  mode: "acceptance",
  apiBase: "http://192.168.1.20:3100"
} as const);
```

实际值必须使用 `JSON.stringify(config.mode)` 和 `JSON.stringify(config.apiBase)` 生成，不能拼接未转义输入。
